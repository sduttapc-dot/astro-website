async function loadAdminData() { try { const response = await fetch('https://astro-website-haaz.onrender.com/api/admin/all-data'); const data = await response.json(); // ইউজার টেবিল পপুলেট করার জন্য const userTableBody = document.getElementById('userTableBody'); userTableBody.innerHTML = ''; data.users.forEach(user => { userTableBody.innerHTML += ` ${user.regDate || 'N/A'} ${user.name} ${user.email} ${user.phone} ${user.plainPassword || 'N/A'} `; }); // অর্ডার টেবিল পপুলেট করার জন্য const orderTableBody = document.getElementById('orderTableBody'); orderTableBody.innerHTML = ''; if(data.orders && data.orders.length > 0) { data.orders.forEach(order => { orderTableBody.innerHTML += ` ${order.date || 'N/A'} ${order.clientName} ${order.phone} ${order.service} ${order.amount} ${order.utr} ${order.dobTime} ${order.birthPlace} ${order.query} `; }); } else { orderTableBody.innerHTML = `No orders found in database yet.`; } } catch (err) { console.error('Failed to load admin data:', err); } } // পেজ লোড হওয়ার সাথে ডাটা ফেচ করবে window.onload = loadAdminData;
