document.getElementById('orderForm').addEventListener('submit', async function(e) { e.preventDefault(); const orderData = { clientName: document.getElementById('clientName').value, phone: document.getElementById('clientPhone').value, service: document.getElementById('servicePackage').value, amount: document.getElementById('serviceAmount').value, utr: document.getElementById('utrRef').value, dobTime: document.getElementById('dobTime').value, birthPlace: document.getElementById('birthPlace').value, query: document.getElementById('clientQuery').value }; try { const response = await fetch('https://astro-website-haaz.onrender.com/api/orders', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(orderData) }); if (response.ok) { alert('Order submitted successfully!'); window.location.reload(); } else { alert('Failed to submit order.'); } } catch (err) { console.error('Error:', err); alert('Server connection error!'); } });
